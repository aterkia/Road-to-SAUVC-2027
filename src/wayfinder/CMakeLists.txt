cmake_minimum_required(VERSION 3.16)
project(wayfinder)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(aterkia_interfaces REQUIRED)
find_package(rclcpp_action REQUIRED)
add_executable(wayfinder_node src/node.cpp)
target_compile_features(wayfinder_node PUBLIC cxx_std_17)
target_compile_options(wayfinder_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(wayfinder_node rclcpp::rclcpp message_filters::message_filters ${sensor_msgs_TARGETS} rclcpp_action::rclcpp_action ${aterkia_interfaces_TARGETS} ${vision_msgs_TARGETS} ${geometry_msgs_TARGETS} ${std_msgs_TARGETS} ${std_srvs_TARGETS} ${nav_msgs_TARGETS} ${diagnostic_msgs_TARGETS})
install(TARGETS wayfinder_node DESTINATION lib/${PROJECT_NAME})
ament_package()
