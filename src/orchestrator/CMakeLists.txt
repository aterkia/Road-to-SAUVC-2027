cmake_minimum_required(VERSION 3.16)
project(orchestrator)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(aterkia_interfaces REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(orchestrator_node src/node.cpp)
target_include_directories(orchestrator_node PRIVATE include)
target_compile_features(orchestrator_node PUBLIC cxx_std_17)
target_compile_options(orchestrator_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(orchestrator_node rclcpp::rclcpp rclcpp_action::rclcpp_action ${aterkia_interfaces_TARGETS} ${std_srvs_TARGETS} ${std_msgs_TARGETS} ${diagnostic_msgs_TARGETS})
install(TARGETS orchestrator_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_plan test/test_plan.cpp)
  target_include_directories(test_plan PRIVATE include)
endif()
ament_package()
