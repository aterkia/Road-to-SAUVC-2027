cmake_minimum_required(VERSION 3.16)
project(helm)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(helm_node src/node.cpp)
target_compile_features(helm_node PUBLIC cxx_std_17)
target_compile_options(helm_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(helm_node rclcpp::rclcpp ${geometry_msgs_TARGETS} ${std_msgs_TARGETS} ${std_srvs_TARGETS} ${nav_msgs_TARGETS} ${diagnostic_msgs_TARGETS})
install(TARGETS helm_node DESTINATION lib/${PROJECT_NAME})
ament_package()
