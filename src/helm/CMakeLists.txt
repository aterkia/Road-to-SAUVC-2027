cmake_minimum_required(VERSION 3.16)
project(helm)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(aterkia_interfaces REQUIRED)
add_executable(helm_node src/node.cpp)
target_compile_features(helm_node PUBLIC cxx_std_17)
target_compile_options(helm_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(helm_node rclcpp::rclcpp ${geometry_msgs_TARGETS} ${std_msgs_TARGETS} ${std_srvs_TARGETS} ${nav_msgs_TARGETS} ${diagnostic_msgs_TARGETS})
install(TARGETS helm_node DESTINATION lib/${PROJECT_NAME})
add_executable(helm_mavros_monitor src/mavros_monitor.cpp)
target_compile_features(helm_mavros_monitor PUBLIC cxx_std_17)
target_compile_options(helm_mavros_monitor PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(helm_mavros_monitor rclcpp::rclcpp ${aterkia_interfaces_TARGETS} ${mavros_msgs_TARGETS} ${sensor_msgs_TARGETS} ${geometry_msgs_TARGETS} ${std_msgs_TARGETS} ${std_srvs_TARGETS} ${diagnostic_msgs_TARGETS})
install(TARGETS helm_mavros_monitor DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(mavlink REQUIRED)
  add_executable(helm_mavlink_peer test/mavlink_peer.cpp)
  target_compile_features(helm_mavlink_peer PUBLIC cxx_std_17)
  target_include_directories(helm_mavlink_peer SYSTEM PRIVATE ${mavlink_INCLUDE_DIRS})
  install(TARGETS helm_mavlink_peer DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
